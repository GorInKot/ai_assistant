    fileExists = Dir(fullPath) <> ""
    
    ' Если файл уже существует, добавляем номер
    fileCounter = 1
    If fileExists Then
        Do While Dir(fullPath) <> ""
            fullPath = savePath & fileName & "_" & fileCounter & ".xlsx"
            fileCounter = fileCounter + 1
        Loop
    End If
    
    ' Сохраняем новую книгу
    On Error Resume Next
    newWorkbook.SaveAs fullPath, FileFormat:=xlOpenXMLWorkbook
    If Err.Number <> 0 Then
        ' Если не удалось сохранить по указанному пути
        MsgBox "Не удалось сохранить файл по пути: " & fullPath & vbCrLf & _
               "Ошибка: " & Err.Description & vbCrLf & _
               "Попробую сохранить в папку Документы.", vbCritical
        
        ' Пытаемся сохранить в Документы
        fullPath = Environ("USERPROFILE") & "\Documents\" & fileName & ".xlsx"
        newWorkbook.SaveAs fullPath, FileFormat:=xlOpenXMLWorkbook
    End If
    On Error GoTo 0
    
    ' Формируем сообщение
    Dim msgText As String
    msgText = "Новая заявка успешно создана!" & vbCrLf & _
              "Файл сохранен: " & vbCrLf & fullPath
              
    If Not wsБаза Is Nothing And count > 0 Then
        msgText = msgText & vbCrLf & "Перенесено записей из базы: " & count
    End If
    
    MsgBox msgText, vbInformation
    newWorkbook.Activate
End Sub
